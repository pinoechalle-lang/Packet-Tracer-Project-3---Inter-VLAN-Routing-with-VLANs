=========================================================================================
SW-A
=========================================================================================

=============[ Escalate Mode ]=============

enable
configure terminal


=============[ Basic Configurations ]=============

hostname SW-A

banner motd #
--- Switch A: AUTHORIZED PERSONNEL ONLY! ---
#

no ip domain-lookup


=============[ Create VLANs ]=============

vlan 10
 name Guest
exit

vlan 20
 name Reception
exit

vlan 30
 name Staff/Maintenance
exit

vlan 40
 name Security
exit

vlan 50
 name Office
exit

vlan 60
 name Voice 
exit

vlan 100
 name Native
exit

vlan 200
 name Remote_Management
exit

vlan 800
 name UNUSED_(SHUTDOWN)
exit


=============[ Assign VLANs to Specific Interface Ranges ]=============

interface range f0/10-11
 description VLAN 10 - Guest
 switchport mode access
 switchport access vlan 10
exit

interface f0/12
 description VLAN 20 - Reception
 switchport mode access
 switchport access vlan 20
exit

interface range f0/13-14
 description VLAN 30 - Staff/Maintenance
 switchport mode access
 switchport access vlan 30
exit


=========[ Assign VLAN 800 to Unused Ports and Shutdown These ]=========

interface range f0/2-9 
 description UNUSED - SHUTDOWN
 switchport mode access 
 switchport access vlan 800
 shutdown
exit

interface range f0/15-24
 description UNUSED - SHUTDOWN
 switchport mode access 
 switchport access vlan 800
 shutdown
exit

interface range g0/1-2
 description UNUSED - SHUTDOWN
 switchport mode access 
 switchport access vlan 800
 shutdown
exit


=============[ Configure Trunking ]=============

interface f0/1
 description Trunk to SW-B
 switchport mode trunk
 switchport trunk native vlan 100
 switchport trunk allowed vlan 10,20,30,40,50,60,100,200
exit


=============[ Remote Management ]=============

interface vlan 200
 description SW-A - Remote Management
 ip address 192.168.200.1 255.255.255.0
 no shutdown
exit

ip default-gateway 192.168.200.254


=============[ Save Running-Configuration (Global) ]=============

do copy running-conf startup-conf
